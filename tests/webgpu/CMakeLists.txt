cmake_minimum_required(VERSION 3.21)
list(APPEND VCPKG_OVERLAY_PORTS "${CMAKE_CURRENT_LIST_DIR}/../../templates/default/tooling/ports")
project(bitloop_webgpu_tests LANGUAGES CXX)

if(NOT TARGET bitloop::webgpu)
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/bitloopWebGPU.cmake")
    bitloop_add_webgpu()
endif()

add_executable(bitloop_webgpu_test compute.cpp)
target_link_libraries(bitloop_webgpu_test PRIVATE bitloop::webgpu)
bitloop_deploy_webgpu_runtime(bitloop_webgpu_test)
install(TARGETS bitloop_webgpu_test RUNTIME DESTINATION bin)
if(BITLOOP_WEBGPU_ASYNC_WAIT)
    target_compile_definitions(bitloop_webgpu_test PRIVATE BITLOOP_TEST_COOPERATIVE_WAIT=1)
endif()
if(EMSCRIPTEN)
    target_link_options(bitloop_webgpu_test PRIVATE -sPTHREAD_POOL_SIZE=1 -sALLOW_MEMORY_GROWTH=1 -sEXIT_RUNTIME=1)
    set_target_properties(bitloop_webgpu_test PROPERTIES SUFFIX ".html")
else()
    enable_testing()
    add_test(NAME bitloop_webgpu_compute COMMAND bitloop_webgpu_test)
    add_executable(bitloop_webgpu_cache_test cache.cpp)
    target_compile_features(bitloop_webgpu_cache_test PRIVATE cxx_std_20)
    add_test(NAME bitloop_webgpu_cache COMMAND bitloop_webgpu_cache_test)
    add_executable(bitloop_webgpu_subgroups_test subgroups.cpp)
    target_link_libraries(bitloop_webgpu_subgroups_test PRIVATE bitloop::webgpu)
    bitloop_deploy_webgpu_runtime(bitloop_webgpu_subgroups_test)
    install(TARGETS bitloop_webgpu_subgroups_test RUNTIME DESTINATION bin)
    add_test(NAME bitloop_webgpu_subgroups COMMAND bitloop_webgpu_subgroups_test)
    set_tests_properties(bitloop_webgpu_subgroups PROPERTIES SKIP_RETURN_CODE 77)
endif()
